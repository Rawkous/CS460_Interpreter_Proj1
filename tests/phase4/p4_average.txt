count = int(input())
values = []
i = 0

while i < count:
    fields = split(input())
    append(values, float(fields[0]))
    i = i + 1

if len(values) == 0:
    print "no values"
else:
    i = 0
    total = 0.0
    while i < len(values):
        total = total + values[i]
        i = i + 1
    print total / len(values)
